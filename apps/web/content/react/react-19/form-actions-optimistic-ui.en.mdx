## 1. Instant Feedback with `useOptimistic`

```tsx title="Zero-Latency Helpful Counter"
'use client'
import { useOptimistic } from 'react'

export function HelpfulButton({ count, onVote }: { count: number; onVote: () => Promise<void> }) {
  const [optimisticCount, addOptimistic] = useOptimistic(count, (curr, delta: number) => curr + delta)

  return (
    <form
      action={async () => {
        addOptimistic(1)
        await onVote()
      }}
    >
      <button type="submit">Helpful ({optimisticCount})</button>
    </form>
  )
}
```

## 2. Choosing Between Native Form Actions and `React Hook Form`

| Requirement | Native React 19 `<form action>` | `React Hook Form` + `Zod` + `TanStack Query` |
| :--- | :--- | :--- |
| Simple 1–2 field server mutation | **Ideal** (zero extra bundle) | Overkill |
| Complex multi-field live Zod validation | Verbose manual field error mapping | **Ideal** (`zodResolver` + field-level subscriptions) |
| Offline-First IndexedDB cache sync | Not built in | **Ideal** (`onMutate` + persisted query cache) |
